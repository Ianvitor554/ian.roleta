def now_iso():
    return datetime.now(timezone.utc).isoformat(timespec="seconds")

def iso_age_seconds(value):
    if not value:
        return None
    try:
        dt=datetime.fromisoformat(str(value).replace("Z","+00:00"))
        if dt.tzinfo is None:
            dt=dt.replace(tzinfo=timezone.utc)
        return max(0.0,(datetime.now(timezone.utc)-dt.astimezone(timezone.utc)).total_seconds())
    except Exception:
        return None

def clamp(x,a,b):
    return max(a,min(b,x))

def normalize(v):
    v = [max(0.0,float(x)) for x in v]
    s = sum(v)
    return [x/s for x in v] if s > 0 else [UNIFORM_P]*37

def softmax(logits):
    m = max(logits)
    e = [math.exp(clamp(x-m,-40,40)) for x in logits]
    return normalize(e)


def temperature_scale(probs, temperature):
    t=max(0.35,min(3.0,float(temperature or 1.0)))
    if abs(t-1.0)<1e-9:
        return normalize(probs)
    inv=1.0/t
    return normalize([max(1e-12,float(p))**inv for p in probs])

def topk(probs,k):
    return sorted(range(37), key=lambda n: probs[n], reverse=True)[:k]

def quantile(values,q):
    vals=sorted(float(x) for x in values if x is not None and math.isfinite(float(x)))
    if not vals:
        return 0.0
    q=clamp(float(q),0.0,1.0)
    pos=(len(vals)-1)*q
    lo=int(math.floor(pos)); hi=int(math.ceil(pos))
    if lo==hi:
        return vals[lo]
    f=pos-lo
    return vals[lo]*(1-f)+vals[hi]*f

def mass(probs, nums):
    return sum(probs[n] for n in set(nums))

def neighbors(n,r=2):
    i = WHEEL_INDEX[int(n)]
    return [WHEEL[(i+d)%37] for d in range(-r,r+1)]

def wheel_distance(a,b):
    ia, ib = WHEEL_INDEX[int(a)], WHEEL_INDEX[int(b)]
    d = abs(ia-ib)
    return min(d,37-d)

def color(n):

    return "red" if n in RED else "black"

def dozen(n):
    n=int(n)
    return 0 if n==0 else 1+(n-1)//12

def terminal(n):

def sector(n):
    n=int(n)

    if n in TIERS: return "tiers"
    return "orphelins"

def brier(probs, actual):
    return sum((probs[i]-(1.0 if i==actual else 0.0))**2 for i in range(37))

def logloss(probs, actual):
    return -math.log(clamp(probs[actual],1e-12,1.0))

def pnl(coverage, actual, chip):
    cov = list(dict.fromkeys(coverage))
    total = len(cov)*chip
    return round((36*chip-total) if actual in cov else -total,2)

def tv_distance(p, q):
    return 0.5 * sum(abs(float(a)-float(b)) for a,b in zip(p,q))

def regime_analysis(nums):
    """
    Mede mudança entre os últimos 30 giros e o bloco anterior.
    TV=0 -> distribuições parecidas; quanto maior, mais houve mudança.
    """
    if len(nums) < 60:
        return {"tv":0.0, "level":"WARMUP", "stable":False}
    recent = freq_probs(nums[-30:], 30)
    prior_slice = nums[-330:-30] if len(nums) >= 330 else nums[:-30]
    prior = freq_probs(prior_slice, max(20, len(prior_slice)))
    tv = tv_distance(recent, prior)
    if tv >= REGIME_BLOCK_TV:
        level = "MUDANÇA_FORTE"
    elif tv >= REGIME_WARN_TV:
        level = "MUDANDO"
    else:
        level = "ESTÁVEL"
    return {"tv":round(tv,4), "level":level, "stable":tv < REGIME_WARN_TV}

def consensus_for_center(center, models, model_weights=None):
    zone = set(neighbors(center,2))
    support = 0.0
    total_weight = 0.0
    details = {}
    baseline = len(zone)/37
    default_w=1/max(1,len(models))
    for name,p in models.items():
        in_top = center in topk(p,8)
        zone_lift = mass(p, zone) / baseline if baseline else 1.0
        ok = in_top or zone_lift >= 1.05
        w=float((model_weights or {}).get(name,default_w))
        support += w*int(ok); total_weight += w
        details[name] = {
            "support":ok,"weight_pct":round(100*w,2),
            "zone_lift":round(zone_lift,3),"center_top8":in_top,
        }
    ratio = support/max(1e-9,total_weight)
    return ratio, details

class DB:
    def __init__(self,path):
        self.path = Path(path)
        self.lock = threading.RLock()
        self.path.parent.mkdir(parents=True,exist_ok=True)
        self._recovering=False
        self.recovery_state={
            "status":"STARTING",
            "last_action":None,
            "last_error":None,
            "restored_from":None,
            "quarantine_dir":None,
            "checked_at":None,
        }

        # Antes de executar WAL ou DDL, verifica se o arquivo existente é legível.
        if DB_AUTO_RECOVER:
            self.recover_if_needed(startup=True)

        self.init()

    def _basic_conn(self,path=None,readonly=False):
        p=Path(path or self.path)
        if readonly:
            uri=f"file:{p.resolve()}?mode=ro"
            c=sqlite3.connect(uri,uri=True,timeout=15)
        else:
            c=sqlite3.connect(p,timeout=30)
        c.row_factory=sqlite3.Row
        c.execute("PRAGMA busy_timeout=30000")
        return c

    def _validate_path(self,path):
        p=Path(path)
        if not p.exists():
            return True,"missing"
        try:
            c=self._basic_conn(p,readonly=True)
            try:
                row=c.execute("PRAGMA quick_check").fetchone()
                result=(row[0] if row else "unknown")
                return result=="ok",str(result)
            finally:
                c.close()
        except Exception as exc:
            return False,f"{type(exc).__name__}: {exc}"

    def _quarantine_current(self,reason):
        stamp=datetime.now().strftime("%Y%m%d_%H%M%S")
        qdir=DB_QUARANTINE_DIR/f"db_corrupt_{stamp}"
        qdir.mkdir(parents=True,exist_ok=True)

        moved=[]
        for suffix in ("","-wal","-shm"):
            src=Path(str(self.path)+suffix)
            if src.exists():
                dst=qdir/src.name
                try:
                    shutil.move(str(src),str(dst))
                    moved.append(str(dst))
                except Exception:
                    try:
                        shutil.copy2(src,dst)
                        moved.append(str(dst))
                        src.unlink(missing_ok=True)
                    except Exception:
                        pass

        meta=qdir/"RECOVERY_INFO.txt"
        try:
            meta.write_text(
                f"timestamp={now_iso()}\n"
                f"database={self.path}\n"
                f"reason={reason}\n"
                f"files={moved}\n",
                encoding="utf-8"
            )
        except Exception:
            pass

        self.recovery_state["quarantine_dir"]=str(qdir)
        return qdir

    def _healthy_backups(self):
        items=[]
        for p in sorted(BACKUP_DIR.glob("mega_ai_*.sqlite3"),reverse=True):
            ok,msg=self._validate_path(p)
            if ok:
                items.append(p)
            else:
                log.warning("Backup inválido ignorado: %s (%s)",p,msg)
        return items

    def recover_if_needed(self,startup=False,force=False):
        if self._recovering:
            return False

        with self.lock:
            self._recovering=True
            try:
                self.recovery_state["checked_at"]=now_iso()

                if not self.path.exists():
                    self.recovery_state.update({
                        "status":"NEW_DB",
                        "last_action":"database file did not exist",
                        "last_error":None,
                    })
                    return False

                ok,msg=self._validate_path(self.path)
                if ok and not force:
                    self.recovery_state.update({
                        "status":"OK",
                        "last_action":"quick_check ok",
                        "last_error":None,
                    })
                    return False

                if not ok:
                    reason=msg
                else:
                    reason="forced recovery"

                self.recovery_state.update({
                    "status":"RECOVERING",
                    "last_error":reason,
                })
                log.error("SQLite inválido detectado: %s",reason)

                qdir=self._quarantine_current(reason)

                # Primeiro tenta restaurar o backup mais novo que esteja íntegro.
                healthy=self._healthy_backups()
                if healthy:
                    backup=healthy[0]
                    shutil.copy2(backup,self.path)

                    # Remove qualquer sidecar residual.
                    for suffix in ("-wal","-shm"):
                        Path(str(self.path)+suffix).unlink(missing_ok=True)

                    ok2,msg2=self._validate_path(self.path)
                    if ok2:
                        self.recovery_state.update({
                            "status":"RESTORED_BACKUP",
                            "last_action":"restored healthy backup",
                            "last_error":None,
                            "restored_from":str(backup),
                            "quarantine_dir":str(qdir),
                        })
                        log.warning("SQLite restaurado automaticamente de %s",backup)
                        return True

                    log.error("Backup restaurado também falhou: %s",msg2)
                    try:
                        self.path.unlink(missing_ok=True)
                    except Exception:
                        pass

                # Nenhum backup íntegro disponível: cria DB novo.
                for suffix in ("","-wal","-shm"):
                    Path(str(self.path)+suffix).unlink(missing_ok=True)

                self.recovery_state.update({
                    "status":"FRESH_DB",
                    "last_action":"created fresh database; corrupt copy preserved",
                    "restored_from":None,
                    "quarantine_dir":str(qdir),
                })
                log.error(
                    "Nenhum backup íntegro encontrado. Banco novo será criado. "
                    "Arquivo corrompido preservado em %s",qdir
                )
                return True

            finally:
                self._recovering=False

    def _open_configured(self):
        c=self._basic_conn(self.path,readonly=False)
        c.execute("PRAGMA journal_mode=WAL")
        c.execute("PRAGMA synchronous=NORMAL")
        c.execute("PRAGMA busy_timeout=30000")
        return c

    def conn(self):
        try:
            return self._open_configured()
        except sqlite3.DatabaseError as exc:
            msg=str(exc).lower()
            corrupt=(
                "malformed" in msg
                or "disk image" in msg
                or "file is not a database" in msg
                or "database corruption" in msg
            )
            if not (DB_AUTO_RECOVER and corrupt):
                raise

            log.error("Erro SQLite em runtime: %s. Iniciando recuperação.",exc)
            self.recover_if_needed(force=True)

            # Um DB novo precisa da estrutura antes da consulta original.
            self.init()
            return self._open_configured()

    def init(self):
        with self.lock, self.conn() as c:
            c.executescript("""
            CREATE TABLE IF NOT EXISTS spins(
                round_id TEXT PRIMARY KEY,
                number INTEGER NOT NULL,
                settled_at TEXT,
                inserted_at TEXT NOT NULL,
                lucky_json TEXT,
                table_name TEXT,
                game_type TEXT
            );

            CREATE TABLE IF NOT EXISTS predictions(
                source_round_id TEXT PRIMARY KEY,
                source_number INTEGER NOT NULL,
                predicted_at TEXT NOT NULL,
                signal TEXT NOT NULL,
                quality REAL NOT NULL,
                threshold REAL NOT NULL,
                reason TEXT,
                centers_json TEXT NOT NULL,
                zones_json TEXT NOT NULL,
                coverage_json TEXT NOT NULL,
                center_details_json TEXT NOT NULL,
                probs_json TEXT NOT NULL,
                features_json TEXT NOT NULL,
                weights_json TEXT NOT NULL,
                ai_metrics_json TEXT NOT NULL,
                next_round_id TEXT,
                actual INTEGER,
                resolved_at TEXT,
                hit INTEGER,
                brier REAL,
                log_loss REAL,
                paper_pnl REAL,
                explanation_json TEXT,
                round_analysis_json TEXT
            );

            CREATE TABLE IF NOT EXISTS state(
                key TEXT PRIMARY KEY,
                value_json TEXT NOT NULL,
                updated_at TEXT NOT NULL
            );
            """)
            cols={r[1] for r in c.execute("PRAGMA table_info(spins)").fetchall()}
            for name,ctype in {
                "lucky_json":"TEXT",
                "table_name":"TEXT",
                "game_type":"TEXT",
            }.items():
                if name not in cols:
                    c.execute(f"ALTER TABLE spins ADD COLUMN {name} {ctype}")
            pcols={r[1] for r in c.execute("PRAGMA table_info(predictions)").fetchall()}
            if "round_analysis_json" not in pcols:
                c.execute("ALTER TABLE predictions ADD COLUMN round_analysis_json TEXT")
            if "model_version" not in pcols:
                c.execute("ALTER TABLE predictions ADD COLUMN model_version TEXT")
            c.commit()

    def number_stats(self):
        with self.lock,self.conn() as c:
            rows=c.execute("SELECT rowid,number,settled_at FROM spins ORDER BY rowid ASC").fetchall()
        total=len(rows)
        positions=[[] for _ in range(37)]
        last_time=[None]*37
        nums=[]
        for idx,r in enumerate(rows):
            n=int(r["number"]); nums.append(n); positions[n].append(idx); last_time[n]=r["settled_at"]
        windows=(20,50,100,300)
        recent_counts={w:[0]*37 for w in windows}
        for w in windows:
            for n in nums[-w:]: recent_counts[w][n]+=1
        expected=total/37 if total else 0.0
        stats=[]
        for n in range(37):
            pos=positions[n]; count=len(pos)
            current_gap=(total-1-pos[-1]) if pos else total
            gaps=[pos[i]-pos[i-1]-1 for i in range(1,len(pos))]
            max_gap=max(gaps+[current_gap]) if (gaps or total) else 0
            mean_gap=(sum(gaps)/len(gaps)) if gaps else None
            z=(count-expected)/math.sqrt(max(expected*(36/37),1e-9)) if expected else 0.0
            stats.append({
                "number":n,"count":count,"pct":100*count/total if total else 0.0,
                "expected":expected,"z":z,"current_gap":current_gap,"max_gap":max_gap,
                "mean_gap":mean_gap,"last_seen":last_time[n],
                "count20":recent_counts[20][n],"count50":recent_counts[50][n],
                "count100":recent_counts[100][n],"count300":recent_counts[300][n],
                "wheel_zone":neighbors(n,2),"color":color(n)
            })
        return {
            "total":total,"expected_per_number":expected,"stats":stats,
            "most_common":sorted(stats,key=lambda x:(x["count"],-x["number"]),reverse=True)[:8],
            "most_delayed":sorted(stats,key=lambda x:(x["current_gap"],x["count"]),reverse=True)[:8],
            "note":"Contagem descritiva do histórico coletado; atraso/frequência não alteram a chance física de uma roleta justa."
        }

    def resolve(self,source_rid,next_rid,actual,explanation):
        with self.lock, self.conn() as c:
            r = c.execute("""
            SELECT coverage_json,probs_json FROM predictions
            WHERE source_round_id=? AND resolved_at IS NULL
            """,(str(source_rid),)).fetchone()
            if not r: return False
            coverage = json.loads(r["coverage_json"])
            probs = json.loads(r["probs_json"])
            hit = int(actual in coverage)
            c.execute("""
            UPDATE predictions SET next_round_id=?,actual=?,resolved_at=?,hit=?,
            brier=?,log_loss=?,paper_pnl=?,explanation_json=?
            WHERE source_round_id=?
            """,(
                str(next_rid),int(actual),now_iso(),hit,brier(probs,actual),
                logloss(probs,actual),pnl(coverage,actual,MIN_CHIP),
                json.dumps(explanation),str(source_rid)
            ))
            c.commit()
            return True

    def metrics(self,limit=1000):
        with self.lock,self.conn() as c:
            rows=c.execute("""
            SELECT signal,hit,paper_pnl,coverage_json,brier,log_loss
            FROM predictions WHERE resolved_at IS NOT NULL
            ORDER BY rowid DESC LIMIT ?
            """,(limit,)).fetchall()
        if not rows:
            return {"resolved":0,"coverage_hit_pct":0,"baseline_pct":0,"green_count":0,
                    "green_hit_pct":0,"green_baseline_pct":0,"green_pnl":0,
                    "green_loss_streak":0,"avg_brier":None,"avg_log_loss":None}
        hits=sum(int(r["hit"] or 0) for r in rows)
        baselines=[len(set(json.loads(r["coverage_json"])))/37 for r in rows]
        greens=[r for r in rows if str(r["signal"]).startswith("GREEN")]
        gh=sum(int(r["hit"] or 0) for r in greens)
        gb=[len(set(json.loads(r["coverage_json"])))/37 for r in greens]
        streak=0
        for r in greens:
            if r["hit"]: break
            streak+=1
        return {
            "resolved":len(rows),
            "coverage_hit_pct":100*hits/len(rows),
            "baseline_pct":100*sum(baselines)/len(baselines),
            "green_count":len(greens),
            "green_hit_pct":100*gh/len(greens) if greens else 0,
            "green_baseline_pct":100*sum(gb)/len(gb) if gb else 0,
            "green_pnl":round(sum(float(r["paper_pnl"] or 0) for r in greens),2),
            "green_loss_streak":streak,
            "avg_brier":sum(float(r["brier"]) for r in rows)/len(rows),
            "avg_log_loss":sum(float(r["log_loss"]) for r in rows)/len(rows),
        }

    def calibration(self,limit=600):
        with self.lock,self.conn() as c:
            rows=c.execute("""
            SELECT probs_json,coverage_json,hit
            FROM predictions
            WHERE resolved_at IS NOT NULL
            ORDER BY rowid DESC LIMIT ?
            """,(limit,)).fetchall()

        if not rows:
            return {"n":0,"expected_pct":0.0,"observed_pct":0.0,"gap_pp":0.0,"ece":0.0}

        bins = [[] for _ in range(5)]
        expected = []
        observed = []
        for r in rows:
            probs=json.loads(r["probs_json"])
            coverage=json.loads(r["coverage_json"])
            p=mass(probs,coverage)
            y=float(r["hit"] or 0)
            expected.append(p); observed.append(y)
            idx=min(4,int(p*5))
            bins[idx].append((p,y))

        ece=0.0
        n=len(rows)
        for b in bins:
            if not b: continue
            ep=sum(x[0] for x in b)/len(b)
            op=sum(x[1] for x in b)/len(b)
            ece += (len(b)/n)*abs(ep-op)

        exp=sum(expected)/n
        obs=sum(observed)/n
        return {
            "n":n,
            "expected_pct":100*exp,
            "observed_pct":100*obs,
            "gap_pp":100*(obs-exp),
            "ece":100*ece,
        }

    def professional_edge_evidence(self,limit=PRO_EVIDENCE_LIMIT,model_version=None):
        def fetch(version):
            with self.lock,self.conn() as c:
                if version:
                    return c.execute("""
                        SELECT signal,coverage_json,hit,paper_pnl
                        FROM predictions
                        WHERE resolved_at IS NOT NULL AND hit IS NOT NULL AND model_version=?
                        ORDER BY rowid DESC LIMIT ?
                    """,(version,int(limit))).fetchall()
                return c.execute("""
                    SELECT signal,coverage_json,hit,paper_pnl
                    FROM predictions
                    WHERE resolved_at IS NOT NULL AND hit IS NOT NULL
                    ORDER BY rowid DESC LIMIT ?
                """,(int(limit),)).fetchall()

        rows=fetch(model_version) if model_version else fetch(None)
        scope="current_version" if model_version else "recent_all_versions"
        if model_version and len(rows)<30:
            rows=fetch(None)
            scope="recent_all_versions_fallback"

        def summarize(items):
            if not items:
                return {"n":0,"hits":0,"observed_pct":0.0,"expected_pct":0.0,
                        "edge_pp":0.0,"z_score":0.0,"paper_pnl":0.0}
            obs=exp=var=pnl_total=0.0
            for r in items:
                try:
                    cov=list(dict.fromkeys(json.loads(r["coverage_json"] or "[]")))
                except Exception:
                    cov=[]
                p=clamp(len(cov)/37,1e-9,1-1e-9)
                y=float(r["hit"] or 0)
                obs+=y
                exp+=p
                var+=p*(1-p)
                pnl_total+=float(r["paper_pnl"] or 0.0)
            n=len(items)
            return {
                "n":n,
                "hits":int(obs),
                "observed_pct":100*obs/n,
                "expected_pct":100*exp/n,
                "edge_pp":100*(obs-exp)/n,
                "z_score":(obs-exp)/math.sqrt(max(var,1e-9)),
                "paper_pnl":round(pnl_total,2),
            }

        all_rows=list(rows)
        greens=[r for r in all_rows if str(r["signal"] or "").startswith("GREEN")]
        return {"scope":scope,"limit":int(limit),
                "all":summarize(all_rows),"green":summarize(greens)}

    def learning_curve(self,limit=600,bin_size=PRO_LEARNING_BIN):
        with self.lock,self.conn() as c:
            rows=c.execute("""
                SELECT log_loss,brier,hit,coverage_json,model_version
                FROM predictions
                WHERE resolved_at IS NOT NULL AND log_loss IS NOT NULL
                ORDER BY rowid DESC LIMIT ?
            """,(int(limit),)).fetchall()
        rows=list(reversed(rows))
        bins=[]
        bs=max(10,int(bin_size))
        for start in range(0,len(rows),bs):
            block=rows[start:start+bs]
            if len(block)<max(10,bs//2):
                continue
            ll=[float(r["log_loss"]) for r in block]
            br=[float(r["brier"]) for r in block if r["brier"] is not None]
            hits=sum(int(r["hit"] or 0) for r in block)
            bases=[]
            for r in block:
                try:
                    cov=list(dict.fromkeys(json.loads(r["coverage_json"] or "[]")))
                    bases.append(len(cov)/37)
                except Exception:
                    pass
            bins.append({
                "n":len(block),
                "avg_log_loss":sum(ll)/len(ll),
                "avg_brier":sum(br)/len(br) if br else None,
                "hit_pct":100*hits/len(block),
                "baseline_pct":100*sum(bases)/len(bases) if bases else 0.0,
                "model_version":block[-1]["model_version"],
            })
        trend=(bins[0]["avg_log_loss"]-bins[-1]["avg_log_loss"]) if len(bins)>=2 else None
        return {"bins":bins[-12:],"trend_log_loss_improvement":trend,
                "improving":bool(trend is not None and trend>0.01),
                "uniform_log_loss":UNIFORM_LOG_LOSS}

    def temperature_calibration(self,limit=TEMP_CALIBRATION_LIMIT,model_version=None):
        with self.lock,self.conn() as c:
            if model_version:
                rows=c.execute("""
                    SELECT probs_json,actual FROM predictions
                    WHERE resolved_at IS NOT NULL AND actual IS NOT NULL AND model_version=?
                    ORDER BY rowid DESC LIMIT ?
                """,(model_version,int(limit))).fetchall()
            else:
                rows=c.execute("""
                    SELECT probs_json,actual FROM predictions
                    WHERE resolved_at IS NOT NULL AND actual IS NOT NULL
                    ORDER BY rowid DESC LIMIT ?
                """,(int(limit),)).fetchall()
        n=len(rows)
        if n<TEMP_CALIBRATION_MIN:
            return {"n":n,"ready":False,"temperature":1.0,"raw_log_loss":None,"best_log_loss":None,"improvement":0.0}
        samples=[]
        for r in rows:
            try: samples.append((json.loads(r["probs_json"]),int(r["actual"])))
            except Exception: pass
        if len(samples)<TEMP_CALIBRATION_MIN:
            return {"n":len(samples),"ready":False,"temperature":1.0,"raw_log_loss":None,"best_log_loss":None,"improvement":0.0}
        candidates=[0.70,0.80,0.90,1.00,1.10,1.20,1.35,1.50,1.70,2.00]
        def avg_loss(t):
            return sum(logloss(temperature_scale(p,t),a) for p,a in samples)/len(samples)
        scored=[(avg_loss(t),t) for t in candidates]
        best_loss,best_t=min(scored)
        raw=next(x[0] for x in scored if abs(x[1]-1.0)<1e-9)
        # Só aplica se houver ganho mensurável; senão mantém 1.0.
        use_t=best_t if best_loss < raw-0.002 else 1.0
        use_loss=best_loss if use_t!=1.0 else raw
        return {"n":len(samples),"ready":True,"temperature":use_t,"raw_log_loss":raw,
                "best_log_loss":use_loss,"improvement":raw-use_loss}

    def backup(self):
        stamp=datetime.now().strftime("%Y%m%d_%H%M%S")
        target=BACKUP_DIR/f"mega_ai_{stamp}.sqlite3"

        with self.lock:
            # Reduz WAL acumulado antes do backup.
            try:
                self.checkpoint("PASSIVE")
            except Exception:
                pass

            src=self.conn()
            dst=sqlite3.connect(target)
            try:
                src.backup(dst)
                dst.commit()
            finally:
                dst.close()
                src.close()

        ok,msg=self._validate_path(target)
        if not ok:
            target.unlink(missing_ok=True)
            raise sqlite3.DatabaseError(f"backup criado mas inválido: {msg}")

        backups=sorted(BACKUP_DIR.glob("mega_ai_*.sqlite3"),reverse=True)
        for old in backups[BACKUP_KEEP:]:
            try:
                old.unlink()
            except Exception:
                pass

        return str(target)

    def collector_state(self):
        return self.get_state("collector_state", {}) or {}

    def save_collector_state(self, value):
        self.set_state("collector_state", value)

def freq_probs(nums,window):
    counts=[1.0]*37
    for n in nums[-window:]:
        counts[int(n)]+=1
    return normalize(counts)

def trans_probs(nums,window=600):
    if not nums: return [UNIFORM_P]*37
    last=int(nums[-1])
    counts=[1.0]*37
    seq=nums[-(window+1):]
    for a,b in zip(seq[:-1],seq[1:]):
        if int(a)==last: counts[int(b)]+=1
    return normalize(counts)


def global_empirical_probs(nums, window=3000):
    data=nums[-window:] if nums else []
    counts=[5.0]*37
    for n in data: counts[int(n)]+=1.0
    return normalize(counts)


def puller_model(nums, source=None):
    """Associações source -> target com Bayes, recência e contexto de 2 passos."""
    if not nums:
        return [UNIFORM_P]*37,{"source":None,"source_support":0,"pair_support":0,"reliability":0.0,"top":[]}

    source=int(nums[-1] if source is None else source)
    baseline=global_empirical_probs(nums,min(PULLER_LONG_WINDOW,max(37,len(nums))))
    alpha=PULLER_PRIOR_ALPHA

    seq=nums[-(PULLER_LONG_WINDOW+1):]
    long_counts=[0.0]*37; source_support=0
    for a,b in zip(seq[:-1],seq[1:]):
        if int(a)==source:
            long_counts[int(b)]+=1.0; source_support+=1
    long_den=source_support+37.0*alpha
    long_p=[(long_counts[t]+alpha)/long_den if long_den>0 else UNIFORM_P for t in range(37)]

    recent=nums[-(PULLER_RECENT_WINDOW+1):]
    recent_counts=[0.0]*37; recent_support=0.0
    for age,(a,b) in enumerate(reversed(list(zip(recent[:-1],recent[1:])))):
        if int(a)!=source: continue
        w=PULLER_RECENCY_DECAY**age
        recent_counts[int(b)]+=w; recent_support+=w
    recent_den=recent_support+37.0*alpha
    recent_p=[(recent_counts[t]+alpha)/recent_den if recent_den>0 else UNIFORM_P for t in range(37)]

    previous=int(nums[-2]) if len(nums)>=2 else None
    pair_counts=[0.0]*37; pair_support=0
    if previous is not None:
        pseq=nums[-(PULLER_PAIR_WINDOW+2):]
        for i in range(len(pseq)-2):
            if int(pseq[i])==previous and int(pseq[i+1])==source:
                pair_counts[int(pseq[i+2])]+=1.0; pair_support+=1
    pair_den=pair_support+37.0*alpha
    pair_p=[(pair_counts[t]+alpha)/pair_den if pair_den>0 else UNIFORM_P for t in range(37)]

    long_conf=1.0-math.exp(-source_support/12.0)
    recent_conf=1.0-math.exp(-recent_support/7.0)
    pair_conf=1.0-math.exp(-pair_support/5.0)
    reliability=clamp(0.62*long_conf+0.25*recent_conf+0.13*pair_conf,0,1)

    raw=[]; details=[]
    for t in range(37):
        base=max(baseline[t],1e-9)
        ll=long_p[t]/base; rl=recent_p[t]/base; pl=pair_p[t]/base
        evidence=(0.56*long_conf*math.log(clamp(ll,0.35,3.5))+
                  0.29*recent_conf*math.log(clamp(rl,0.35,4.0))+
                  0.15*pair_conf*math.log(clamp(pl,0.35,4.5)))
        raw.append(base*math.exp(clamp(evidence,-0.70,0.70)))
        expected=max(source_support*base,1e-6)
        z=(long_counts[t]-expected)/math.sqrt(max(expected*(1-base),0.25))
        strength=100*clamp(0.34*clamp((ll-1)/1.2,0,1)+0.24*clamp((rl-1)/1.4,0,1)+
                           0.14*clamp((pl-1)/1.6,0,1)+0.18*long_conf+0.10*clamp((z-.5)/2.5,0,1),0,1)
        details.append({"target":t,"count":int(long_counts[t]),"source_support":source_support,
                        "recent_weighted_count":round(recent_counts[t],3),"recent_support":round(recent_support,3),
                        "pair_count":int(pair_counts[t]),"pair_support":pair_support,
                        "posterior_pct":round(100*long_p[t],3),"long_lift":round(ll,3),
                        "recent_lift":round(rl,3),"pair_lift":round(pl,3),"z":round(z,3),
                        "strength":round(strength,1),"zone":neighbors(t,2)})
    probs=normalize(raw)
    for d in details: d["model_probability_pct"]=round(100*probs[d["target"]],3)
    details.sort(key=lambda d:(d["strength"],d["model_probability_pct"],d["count"]),reverse=True)
    shown=[d for d in details[:10] if d["strength"]>=PULLER_MIN_DISPLAY_STRENGTH] or details[:5]
    return probs,{"source":source,"previous":previous,"source_support":source_support,
                  "recent_support":round(recent_support,3),"pair_support":pair_support,
                  "reliability":round(reliability,4),"enough_support":source_support>=PULLER_MIN_SOURCE_SUPPORT,
                  "strong_support":source_support>=PULLER_STRONG_SOURCE_SUPPORT,"top":shown}


def strongest_puller_relations(nums, limit=12):
    if len(nums)<80: return []
    rows=[]
    for source in range(37):
        seq=list(nums)
        if not seq or seq[-1]!=source: seq=seq+[source]
        _,rep=puller_model(seq,source=source)
        if rep["source_support"]<PULLER_MIN_SOURCE_SUPPORT: continue
        for item in rep["top"][:3]:
            rows.append({"source":source,"target":item["target"],"strength":item["strength"],
                         "lift":item["long_lift"],"count":item["count"],"support":rep["source_support"]})
    rows.sort(key=lambda x:(x["strength"],x["count"],x["lift"]),reverse=True)
    return rows[:limit]


def wilson_lower(hits,n,z=1.2816):
    if n<=0: return 0.0
    p=hits/n
    den=1+(z*z/n)
    center=p+(z*z/(2*n))
    margin=z*math.sqrt((p*(1-p)+z*z/(4*n))/n)
    return max(0.0,(center-margin)/den)


def source_local_validation(nums,source=None,max_events=None):
    """Replay walk-forward apenas quando o MESMO número-fonte apareceu."""
    if not nums:
        return {"source":None,"n":0,"top1_pct":0.0,"top3_pct":0.0,"top5_pct":0.0,
                "top5_lower_pct":0.0,"avg_log_loss":None,"skill":0.0,
                "enough_data":False,"recent_top5_pct":0.0}

    source=int(nums[-1] if source is None else source)
    max_events=max_events or ROUND_LOCAL_MAX_EVENTS
    idxs=[i for i in range(45,len(nums)-1) if int(nums[i])==source][-max_events:]

    t1=t3=t5=0
    losses=[]
    recent=[]

    for i in idxs:
        before=nums[:i+1]
        actual=int(nums[i+1])
        p,rep=puller_model(before,source=source)
        if rep.get("source_support",0)<3:
            continue
        rank=topk(p,5)
        h1=int(actual in rank[:1]); h3=int(actual in rank[:3]); h5=int(actual in rank[:5])
        t1+=h1; t3+=h3; t5+=h5
        recent.append(h5)
        losses.append(logloss(p,actual))

    n=len(losses)
    if not n:
        return {"source":source,"n":0,"top1_pct":0.0,"top3_pct":0.0,"top5_pct":0.0,
                "top5_lower_pct":0.0,"avg_log_loss":None,"skill":0.0,
                "enough_data":False,"recent_top5_pct":0.0}

    top5=t5/n
    ll=sum(losses)/n
    ll_skill=clamp((UNIFORM_LOG_LOSS-ll)/0.22,0,1)
    top5_skill=clamp((top5-(5/37))/0.12,0,1)
    shrink=n/(n+18.0)
    skill=shrink*(0.58*ll_skill+0.42*top5_skill)
    rr=recent[-20:]

    return {
        "source":source,"n":n,
        "top1_pct":100*t1/n,"top3_pct":100*t3/n,"top5_pct":100*top5,
        "top5_lower_pct":100*wilson_lower(t5,n),
        "top5_baseline_pct":100*(5/37),
        "recent_top5_pct":100*(sum(rr)/len(rr)) if rr else 0.0,
        "avg_log_loss":ll,"uniform_log_loss":UNIFORM_LOG_LOSS,
        "skill":clamp(skill,0,1),
        "enough_data":n>=ROUND_LOCAL_MIN_EVENTS,
    }


def pair_context_report(nums):
    if len(nums)<2:
        return {"previous":None,"source":None,"support":0,"top":[]}
    previous=int(nums[-2]); source=int(nums[-1])
    counts=[0]*37; support=0
    seq=nums[-(PULLER_PAIR_WINDOW+2):]
    for i in range(len(seq)-2):
        if int(seq[i])==previous and int(seq[i+1])==source:
            counts[int(seq[i+2])]+=1; support+=1
    den=support+37.0
    p=[(c+1.0)/den for c in counts]
    rank=topk(p,6)
    return {"previous":previous,"source":source,"support":support,
            "top":[{"number":n,"count":counts[n],"pct":100*p[n]} for n in rank]}


def cluster_context_report(nums):

    return {"source":source,"zone":neighbors(source,2),"recent_hits":hits,
            "recent_n":len(recent),"recent_pct":100*hits/len(recent) if recent else 0.0,
            "fair_pct":100*len(zone)/37}


def cluster_probs(nums,window=220):
    counts=[0.35]*37
    for age,n in enumerate(reversed(nums[-window:])):
        w=0.995**age
        for x in neighbors(n,2): counts[x]+=w
    return normalize(counts)

def terminal_probs(nums,window=300):
    t=[1.0]*10
    for n in nums[-window:]: t[terminal(n)]+=1
    total=sum(t)
    p=[0.0]*37
    for n in range(37):
        group=[x for x in range(37) if terminal(x)==terminal(n)]
        p[n]=(t[terminal(n)]/total)/len(group)
    return normalize(p)

def sector_probs(nums,window=250):
    groups={"voisins":VOISINS,"tiers":TIERS,"orphelins":ORPHELINS}
    cnt={k:1.0 for k in groups}
    for n in nums[-window:]: cnt[sector(n)]+=1
    total=sum(cnt.values())
    p=[0.0]*37
    for n in range(37):
        s=sector(n)
        p[n]=(cnt[s]/total)/len(groups[s])
    return normalize(p)

def gap(nums,c,cap=180):
    g=0
    for n in reversed(nums[-cap:]):
        if int(n)==c: break
        g+=1
    else: g=cap
    return clamp(g/cap,0,1)

def rel(p):
    return clamp((p/UNIFORM_P-1)/2,-1,1)

def feature_matrix(nums):
    pull_probs,_pull_report=puller_model(nums)
    ps=[
        pull_probs,freq_probs(nums,20),freq_probs(nums,50),

        cluster_probs(nums),terminal_probs(nums),sector_probs(nums)
    ]
    last=int(nums[-1]) if nums else None
    rows=[]
    for c in range(37):
        if last is None:
            near=samet=samec=samed=0
        else:
            near=1-wheel_distance(last,c)/18
            samet=1 if terminal(last)==terminal(c) else 0
            samec=1 if last!=0 and c!=0 and color(last)==color(c) else 0
            samed=1 if dozen(last)!=0 and dozen(last)==dozen(c) else 0
        rows.append([

            rel(ps[4][c]),rel(ps[5][c]),rel(ps[6][c]),rel(ps[7][c]),
            gap(nums,c),near,samet,samec,samed,rel(ps[8][c])
        ])
    return rows

def classical_probs(nums):
    models=[
        (trans_probs(nums),.10),(freq_probs(nums,20),.10),(freq_probs(nums,50),.11),
        (freq_probs(nums,100),.13),(freq_probs(nums,300),.11),(freq_probs(nums,1000),.09),

    out=[0.0]*37

def window_agreement(nums):
    sets=[]
    for w in (20,50,100,300,1000):
        if len(nums)>=20: sets.append(set(topk(freq_probs(nums,min(w,len(nums))),8)))
    if len(sets)<2: return 0
    vals=[]
    for i in range(len(sets)):
        for j in range(i+1,len(sets)):
            vals.append(len(sets[i]&sets[j])/max(1,len(sets[i]|sets[j])))
    return sum(vals)/len(vals)

class PullerValidator:
    KEY="puller_validator_v1"
    def __init__(self,db):
        self.db=db; st=db.get_state(self.KEY,{}) or {}
        self.samples=int(st.get("samples",0)); self.t1=int(st.get("t1",0)); self.t3=int(st.get("t3",0)); self.t5=int(st.get("t5",0))
        self.loss=float(st.get("loss",0.0)); self.bs=float(st.get("brier",0.0)); self.lock=threading.RLock()
        self.bootstrap_samples=int(st.get("bootstrap_samples",0))
    def save(self):
        self.db.set_state(self.KEY,{"samples":self.samples,"t1":self.t1,"t3":self.t3,"t5":self.t5,"loss":self.loss,"brier":self.bs,"bootstrap_samples":self.bootstrap_samples})
    def bootstrap(self,nums):
        """Replay walk-forward, sem futuro, apenas quando ainda não há estado do validador."""
        if self.samples>0 or len(nums)<80:
            return 0
        start=max(60,len(nums)-PULLER_BOOTSTRAP_POINTS)
        before=self.samples
        for i in range(start,len(nums)):
            self.evaluate(nums[:i],int(nums[i]))
        self.bootstrap_samples=self.samples-before
        self.save()
        return self.bootstrap_samples
    def evaluate(self,nums_before,actual):
        if len(nums_before)<30: return
        p,rep=puller_model(nums_before)
        if rep["source_support"]<2: return
        rank=topk(p,5); actual=int(actual)
        with self.lock:
            self.samples+=1; self.t1+=int(actual in rank[:1]); self.t3+=int(actual in rank[:3]); self.t5+=int(actual in rank)
            self.loss+=logloss(p,actual); self.bs+=brier(p,actual)
            if self.samples%5==0: self.save()
    def metrics(self):
        n=self.samples; ll=self.loss/n if n else None
        skill=clamp((UNIFORM_LOG_LOSS-ll)/0.22,0,1) if n>=PULLER_SKILL_MIN_SAMPLES and ll is not None else 0.0
        return {"samples":n,"ready":n>=PULLER_SKILL_MIN_SAMPLES,
                "top1_pct":100*self.t1/n if n else 0.0,"top3_pct":100*self.t3/n if n else 0.0,
                "top5_pct":100*self.t5/n if n else 0.0,"avg_log_loss":ll,"avg_brier":self.bs/n if n else None,
                "skill":skill,"uniform_log_loss":UNIFORM_LOG_LOSS,"bootstrap_samples":self.bootstrap_samples}

class ModelPerformanceTracker:
    KEY="professional_model_tracker_v2"
    OLD_KEY="adaptive_model_tracker_v1"

    def __init__(self,db):
        self.db=db
        st=db.get_state(self.KEY,{}) or {}
        self.stats=st.get("stats",{}) if isinstance(st.get("stats",{}),dict) else {}
        self.regimes=st.get("regimes",{}) if isinstance(st.get("regimes",{}),dict) else {}
        self.lock=threading.RLock()

        if not self.stats:
            old=db.get_state(self.OLD_KEY,{}) or {}
            oldstats=old.get("stats",{}) if isinstance(old.get("stats",{}),dict) else {}
            for name,v in oldstats.items():
                ll=v.get("ema_log_loss")
                br=v.get("ema_brier")
                t5=v.get("ema_top5")
                self.stats[name]={
                    "n":int(v.get("n",0)),
                    "fast_ll":ll,"slow_ll":ll,
                    "fast_brier":br,"slow_brier":br,
                    "fast_top5":t5,"slow_top5":t5,
                }
        for name in MODEL_PRIORS:
            self.stats.setdefault(name,self._blank())

    @staticmethod
    def _blank():
        return {"n":0,"fast_ll":None,"slow_ll":None,
                "fast_brier":None,"slow_brier":None,
                "fast_top5":None,"slow_top5":None}

    def save(self):
        self.db.set_state(self.KEY,{"stats":self.stats,"regimes":self.regimes})

    def _update(self,st,ll,br,h):
        st["n"]=int(st.get("n",0))+1
        for key,val,alpha in (
            ("fast_ll",ll,PRO_MODEL_FAST_ALPHA),
            ("slow_ll",ll,PRO_MODEL_SLOW_ALPHA),
            ("fast_brier",br,PRO_MODEL_FAST_ALPHA),
            ("slow_brier",br,PRO_MODEL_SLOW_ALPHA),
            ("fast_top5",h,PRO_MODEL_FAST_ALPHA),
            ("slow_top5",h,PRO_MODEL_SLOW_ALPHA),
        ):
            old=st.get(key)
            st[key]=float(val) if old is None else (1-alpha)*float(old)+alpha*float(val)

    def evaluate(self,nums_before,actual,ai):
        if len(nums_before)<30:
            return
        ai_probs,_=ai.predict(nums_before)
        models=independent_models(nums_before,ai_probs)
        regime=regime_analysis(nums_before).get("level","WARMUP")
        actual=int(actual)

        with self.lock:
            rb=self.regimes.setdefault(regime,{})
            for name,p in models.items():
                ll=logloss(p,actual)
                br=brier(p,actual)
                h=float(actual in topk(p,5))
                gs=self.stats.setdefault(name,self._blank())
                rs=rb.setdefault(name,self._blank())
                self._update(gs,ll,br,h)
                self._update(rs,ll,br,h)
            if max(int(v.get("n",0)) for v in self.stats.values())%5==0:
                self.save()

    def _combined(self,name,regime=None):
        g=self.stats.get(name,self._blank())
        r=(self.regimes.get(regime,{}) or {}).get(name,self._blank()) if regime else self._blank()

        def mv(field):
            gv=g.get(field)
            rv=r.get(field)
            if gv is None:
                return rv
            if rv is None or not regime:
                return gv
            rc=clamp(int(r.get("n",0))/max(1,PRO_REGIME_MIN_SAMPLES),0,1)
            return (1-0.70*rc)*float(gv)+(0.70*rc)*float(rv)

        return {"n":int(g.get("n",0)),
                "regime_n":int(r.get("n",0)) if regime else 0,
                "fast_ll":mv("fast_ll"),"slow_ll":mv("slow_ll"),
                "fast_brier":mv("fast_brier"),"slow_brier":mv("slow_brier"),
                "fast_top5":mv("fast_top5"),"slow_top5":mv("slow_top5")}

    def weights(self,regime=None):
        scores={}
        for name,prior in MODEL_PRIORS.items():
            st=self._combined(name,regime)
            n=int(st["n"])
            conf=clamp(n/max(1,MODEL_WEIGHT_MIN_SAMPLES),0,1)
            fast,slow=st["fast_ll"],st["slow_ll"]

            if fast is None and slow is None:
                scores[name]=max(PRO_WEIGHT_FLOOR,float(prior))
                continue
            if fast is None: fast=slow
            if slow is None: slow=fast

            mixed=0.65*float(slow)+0.35*float(fast)
            skill=clamp((UNIFORM_LOG_LOSS-mixed)/MODEL_WEIGHT_TAU,-1.25,1.25)
            ft,st5=st["fast_top5"],st["slow_top5"]
            if ft is None and st5 is None:
                top_edge=0.0
            else:
                if ft is None: ft=st5
                if st5 is None: st5=ft
                top_edge=(0.40*float(ft)+0.60*float(st5))-(5/37)

            stability=math.exp(-abs(float(fast)-float(slow))/0.20)
            perf=math.exp(clamp(skill,-1.1,1.1))
            top_adj=1.0+clamp(top_edge/0.10,-0.30,0.30)
            learned=float(prior)*perf*top_adj*(0.75+0.25*stability)
            scores[name]=max(PRO_WEIGHT_FLOOR,float(prior)*(1-conf)+learned*conf)

        total=sum(scores.values()) or 1.0
        return {k:v/total for k,v in scores.items()}

    def metrics(self,regime=None):
        weights=self.weights(regime)
        out={}
        for name in MODEL_PRIORS:
            st=self._combined(name,regime)
            fast,slow=st["fast_ll"],st["slow_ll"]
            if fast is None and slow is None:
                mixed=None
            else:
                if fast is None: fast=slow
                if slow is None: slow=fast
                mixed=0.65*float(slow)+0.35*float(fast)

            out[name]={
                "n":int(st["n"]),"regime_n":int(st["regime_n"]),
                "fast_log_loss":st["fast_ll"],"slow_log_loss":st["slow_ll"],
                "mixed_log_loss":mixed,
                "fast_top5_pct":100*float(st["fast_top5"] or 0),
                "slow_top5_pct":100*float(st["slow_top5"] or 0),
                "weight_pct":100*weights.get(name,0),
                "beats_uniform":bool(mixed is not None and mixed<UNIFORM_LOG_LOSS),
                "stable":bool(st["fast_ll"] is not None and st["slow_ll"] is not None
                              and abs(float(st["fast_ll"])-float(st["slow_ll"]))<0.18),
            }

        mature=sum(1 for v in out.values() if v["n"]>=MODEL_WEIGHT_MIN_SAMPLES)
        beating=sum(1 for v in out.values()
                    if v["n"]>=MODEL_WEIGHT_MIN_SAMPLES and v["beats_uniform"])
        return {"models":out,"weights":weights,"min_samples":MODEL_WEIGHT_MIN_SAMPLES,
                "regime":regime,"mature_models":mature,
                "models_beating_uniform":beating}


def cap_and_renormalize(weights,name,cap):
    weights=dict(weights)
    if weights.get(name,0)<=cap: return weights
    excess=weights[name]-cap; weights[name]=cap
    others=[k for k in weights if k!=name]
    base=sum(weights[k] for k in others)
    if base>0:
        for k in others: weights[k]+=excess*(weights[k]/base)
    return weights

class OnlineAI:
    KEY="online_ai_v2_professional"
    OLD_KEY="online_ai_v1"

    def __init__(self,db):
        self.db=db
        state=db.get_state(self.KEY,{}) or {}
        if not state:
            state=db.get_state(self.OLD_KEY,{}) or {}

        w=state.get("weights")
        if not isinstance(w,list) or len(w)!=len(FEATURES):
            w=[0.0]*len(FEATURES)
            for name,val in {"transicao":.10,"freq50":.05,"freq100":.05,"cluster_roda":.08}.items():
                w[FEATURES.index(name)]=val
        self.weights=[float(x) for x in w]

        gs=state.get("grad_sq")
        if not isinstance(gs,list) or len(gs)!=len(FEATURES):
            gs=[0.0]*len(FEATURES)
        self.grad_sq=[max(0.0,float(x)) for x in gs]

        self.samples=int(state.get("samples",0))
        self.t1=int(state.get("t1",0))
        self.t3=int(state.get("t3",0))
        self.t5=int(state.get("t5",0))
        self.loss=float(state.get("loss",0.0))
        self.bs=float(state.get("brier",0.0))
        self.fast_ll=state.get("fast_ll")
        self.slow_ll=state.get("slow_ll")
        self.fast_top5=state.get("fast_top5")
        self.slow_top5=state.get("slow_top5")
        self.lock=threading.RLock()

    def save(self):
        self.db.set_state(self.KEY,{
            "weights":self.weights,"grad_sq":self.grad_sq,"samples":self.samples,
            "t1":self.t1,"t3":self.t3,"t5":self.t5,"loss":self.loss,"brier":self.bs,
            "fast_ll":self.fast_ll,"slow_ll":self.slow_ll,
            "fast_top5":self.fast_top5,"slow_top5":self.slow_top5})

    def probs_from_matrix(self,matrix,weights=None):
        weights=self.weights if weights is None else weights
        return softmax([sum(w*x for w,x in zip(weights,row)) for row in matrix])

    def predict(self,nums):
        m=feature_matrix(nums)
        return self.probs_from_matrix(m),m

    @staticmethod
    def _ema(old,value,alpha):
        return float(value) if old is None else (1-alpha)*float(old)+alpha*float(value)

    def learn(self,nums_before,actual):
        if len(nums_before)<10:
            return
        actual=int(actual)

        with self.lock:
            m=feature_matrix(nums_before)
            p=self.probs_from_matrix(m)
            rank=topk(p,5)
            ll=logloss(p,actual)
            br=brier(p,actual)
            h5=float(actual in rank)

            self.samples+=1
            self.t1+=int(actual in rank[:1])
            self.t3+=int(actual in rank[:3])
            self.t5+=int(actual in rank)
            self.loss+=ll
            self.bs+=br

            self.fast_ll=self._ema(self.fast_ll,ll,AI_FAST_ALPHA)
            self.slow_ll=self._ema(self.slow_ll,ll,AI_SLOW_ALPHA)
            self.fast_top5=self._ema(self.fast_top5,h5,AI_FAST_ALPHA)
            self.slow_top5=self._ema(self.slow_top5,h5,AI_SLOW_ALPHA)

            expected=[0.0]*len(self.weights)
            mean_feature=[0.0]*len(self.weights)
            for c in range(37):
                for j,x in enumerate(m[c]):
                    expected[j]+=p[c]*x
                    mean_feature[j]+=x/37.0

            smooth=AI_LABEL_SMOOTHING
            target=[(1-smooth)*m[actual][j]+smooth*mean_feature[j]
                    for j in range(len(self.weights))]

            for j in range(len(self.weights)):
                grad=expected[j]-target[j]+AI_L2*self.weights[j]
                grad=clamp(grad,-AI_GRAD_CLIP,AI_GRAD_CLIP)
                self.grad_sq[j]+=grad*grad
                step=AI_LR/math.sqrt(self.grad_sq[j]+AI_ADAGRAD_EPS)
                self.weights[j]=clamp(self.weights[j]-step*grad,
                                      -AI_WEIGHT_CLIP,AI_WEIGHT_CLIP)

            if self.samples%5==0:
                self.save()

    def metrics(self):
        n=self.samples
        avg_ll=self.loss/n if n else None
        fast,slow=self.fast_ll,self.slow_ll
        if fast is None and slow is None:
            pro_ll=avg_ll
        else:
            if fast is None: fast=slow
            if slow is None: slow=fast
            pro_ll=0.35*float(fast)+0.65*float(slow)

        conf=clamp(n/max(1,AI_SKILL_MIN_SAMPLES),0,1)
        skill=conf*clamp((UNIFORM_LOG_LOSS-float(pro_ll))/0.22,0,1) if pro_ll is not None else 0.0

        return {
            "samples":n,"ready":n>=AI_MIN_SAMPLES,
            "top1_pct":100*self.t1/n if n else 0,
            "top3_pct":100*self.t3/n if n else 0,
            "top5_pct":100*self.t5/n if n else 0,
            "avg_log_loss":avg_ll,"avg_brier":self.bs/n if n else None,
            "fast_log_loss":self.fast_ll,"slow_log_loss":self.slow_ll,
            "fast_top5_pct":100*float(self.fast_top5 or 0),
            "slow_top5_pct":100*float(self.slow_top5 or 0),
            "professional_skill":skill,
            "weights":{FEATURES[i]:round(w,4) for i,w in enumerate(self.weights)}
        }

    @staticmethod
    def explain(row,weights,topn=5):
        items=[]
        for name,x,w in zip(FEATURES,row,weights):
            if name=="bias":
                continue
            c=float(x)*float(w)
            items.append({"factor":name,"contribution":round(c,4),
                          "direction":"favorece" if c>0 else "reduz" if c<0 else "neutro"})
        items.sort(key=lambda z:abs(z["contribution"]),reverse=True)
        return items[:topn]

def choose_centers(probs,nums,ai,matrix,model_weights=None):
    """
    V4.0: otimização combinatória dos 3 centros.

    Em vez de escolher centro por centro, testa todas as combinações entre os
    melhores candidatos e avalia a cobertura conjunta. Com 14 candidatos são
    somente 364 combinações, rápido o suficiente para cada rodada.
    """
    agreement=window_agreement(nums)
    ai_probs=ai.probs_from_matrix(matrix)
    models=independent_models(nums,ai_probs)
    pull_probs,pull_report=puller_model(nums)

    ranked=sorted(
        range(37),
        key=lambda n:(mass(probs,neighbors(n,2)),probs[n]),
        reverse=True
    )[:CENTER_COMBO_CANDIDATES]

    info={}
    pull_rank=topk(pull_probs,8)

    for n in ranked:
        zone=neighbors(n,2)
        zp=mass(probs,zone)
        base=len(set(zone))/37
        lift=zp/base if base else 1.0
        factors=OnlineAI.explain(matrix[n],ai.weights,5)
        pos=sum(max(0,f["contribution"]) for f in factors)
        consensus,consensus_details=consensus_for_center(n,models,model_weights)
        pull_zone_lift=mass(pull_probs,zone)/(len(set(zone))/37)
        pull_bonus=clamp(
            0.55*float(n in pull_rank)
            +0.45*clamp((pull_zone_lift-1.0)/0.35,0,1),
            0,1
        )*float(pull_report.get("reliability",0.0))

        score=(
            24*clamp((lift-.95)/.25,0,1)
            +17*clamp(probs[n]/max(probs),0,1)
            +16*clamp(agreement/.70,0,1)
            +13*clamp(pos/.5,0,1)
            +20*consensus
            +10*pull_bonus
        )

        info[n]={
            "center":n,
            "score":round(clamp(score,0,100),1),
            "zone_probability":zp,
            "zone_lift":lift,
            "ai_factors":factors,
            "model_consensus":consensus,
            "model_support":consensus_details,
            "puller_zone_lift":round(pull_zone_lift,3),
            "puller_support":pull_report.get("source_support",0),
            "puller_reliability":pull_report.get("reliability",0.0),
            "pulled_by_last":n in pull_rank,
        }

    best_combo=None
    best_obj=-1e18
    best_coverage=None

    # Avalia o conjunto completo, não só centros isolados.
    for combo in combinations(ranked,3):
        zones=[set(neighbors(n,2)) for n in combo]
        coverage=set().union(*zones)
        overlap=sum(len(zones[i]&zones[j]) for i in range(3) for j in range(i+1,3))
        covp=mass(probs,coverage)
        base=len(coverage)/37
        covlift=covp/base if base else 1.0

        avg_score=sum(info[n]["score"] for n in combo)/3
        avg_cons=sum(info[n]["model_consensus"] for n in combo)/3
        avg_pull=sum(clamp((info[n]["puller_zone_lift"]-1)/.5,0,1) for n in combo)/3

        # Distância física mínima favorece 3 regiões realmente diferentes.
        distances=[
            wheel_distance(combo[0],combo[1]),
            wheel_distance(combo[0],combo[2]),
            wheel_distance(combo[1],combo[2]),
        ]
        spacing=clamp(min(distances)/6.0,0,1)

        obj=(
            150*covp
            +38*clamp((covlift-1.0)/.25,0,1)
            +0.60*avg_score
            +18*avg_cons
            +8*avg_pull
            +8*spacing
            -5.5*overlap
        )

        if obj>best_obj:
            best_obj=obj
            best_combo=combo
            best_coverage=coverage

    if best_combo is None:
        best_combo=tuple(ranked[:3])
        best_coverage=set().union(*(set(neighbors(n,2)) for n in best_combo))

    chosen=list(best_combo)
    # Exibe os centros mais fortes primeiro, mas mantém a combinação ótima.
    chosen.sort(key=lambda n:info[n]["score"],reverse=True)

    zones={str(n):neighbors(n,2) for n in chosen}
    coverage=sorted(best_coverage,key=lambda n:WHEEL_INDEX[n])
    details=[info[n] for n in chosen]

    model_consensus=sum(d["model_consensus"] for d in details)/len(details) if details else 0.0
    return chosen,zones,coverage,details,agreement,model_consensus

def make_signal(probs,nums,ai,matrix,metrics,calibration,model_weights=None):
    centers,zones,coverage,details,agreement,model_consensus=choose_centers(probs,nums,ai,matrix,model_weights)
    covp=mass(probs,coverage)
    covbase=len(coverage)/37
    covlift=covp/covbase if covbase else 1
    top5=topk(probs,5)
    top5lift=mass(probs,top5)/(5/37)
    entropy=-sum(p*math.log(max(p,1e-12)) for p in probs)/math.log(37)
    am=ai.metrics()
    regime=regime_analysis(nums)

    scores=[d["score"] for d in details]
    center_ok=(
        len(scores)==3
        and scores[0]>=MIN_CENTER_1
        and scores[1]>=MIN_CENTER_2
        and scores[2]>=MIN_CENTER_3
    )

    # Calibração só vira gate depois de amostra suficiente.
    calibration_ok=True
    if calibration["n"]>=CALIBRATION_MIN_N:
        calibration_ok=calibration["gap_pp"]>=-5.0 and calibration["ece"]<=12.0

    quality=(
        18*clamp((covlift-1)/.2,0,1)
        +12*clamp((top5lift-1)/.3,0,1)
        +15*clamp(agreement/.7,0,1)
        +16*model_consensus
        +12*clamp((1-entropy)/.2,0,1)
        +15*(sum(scores)/300 if scores else 0)
        +7*clamp(am["samples"]/AI_MIN_SAMPLES,0,1)
        +5*(1-clamp(regime["tv"]/REGIME_BLOCK_TV,0,1))
    )
    quality=round(clamp(quality,0,100),1)

    threshold=GREEN_SCORE+min(8,3*metrics["green_loss_streak"])
    if metrics["green_count"]>=20 and metrics["green_hit_pct"]<metrics["green_baseline_pct"]:
        threshold+=4
    if calibration["n"]>=CALIBRATION_MIN_N and calibration["gap_pp"] < -3:
        threshold+=2
    if regime["level"]=="MUDANDO":
        threshold+=2
    threshold=round(clamp(threshold,GREEN_SCORE,98),1)

    reasons=[]
    if metrics["resolved"]<GREEN_MIN_RESOLVED:
        reasons.append(f"histórico {metrics['resolved']}/{GREEN_MIN_RESOLVED}")
    if am["samples"]<AI_MIN_SAMPLES:
        reasons.append(f"IA aprendendo {am['samples']}/{AI_MIN_SAMPLES}")
    if agreement<MIN_WINDOW_AGREEMENT:
        reasons.append(f"consenso temporal baixo {agreement*100:.0f}%")
    if model_consensus<MIN_MODEL_CONSENSUS:
        reasons.append(f"modelos discordam {model_consensus*100:.0f}%")
    if covlift<1.03:
        reasons.append("cobertura pouco concentrada")
    if top5lift<1.05:
        reasons.append("TOP5 pouco concentrado")
    if not center_ok:
        reasons.append("um dos 3 centros está fraco")
    if entropy>.985:
        reasons.append("distribuição quase uniforme")
    if metrics["green_loss_streak"]>MAX_GREEN_LOSS_STREAK:
        reasons.append("bloqueio por perdas Green")
    if regime["tv"]>=REGIME_BLOCK_TV:
        reasons.append(f"regime mudou forte TV={regime['tv']:.2f}")
    if not calibration_ok:
        reasons.append("calibração recente não valida confiança")

    green=(
        metrics["resolved"]>=GREEN_MIN_RESOLVED
        and am["samples"]>=AI_MIN_SAMPLES
        and agreement>=MIN_WINDOW_AGREEMENT
        and model_consensus>=MIN_MODEL_CONSENSUS
        and covlift>=1.03
        and top5lift>=1.05
        and center_ok
        and entropy<=.985
        and metrics["green_loss_streak"]<=MAX_GREEN_LOSS_STREAK
        and regime["tv"]<REGIME_BLOCK_TV
        and calibration_ok
        and quality>=threshold
    )

    if green:
        if (
            quality>=GREEN_ELITE_SCORE
            and model_consensus>=.82
            and agreement>=.62
            and regime["tv"]<REGIME_WARN_TV
        ):
            signal="GREEN_ELITE"
        elif quality>=GREEN_PLUS_SCORE and model_consensus>=.72:
            signal="GREEN_PLUS"
        else:
            signal="GREEN"
    elif quality>=YELLOW_SCORE and agreement>=.30:
        signal="YELLOW"
    else:
        signal="RED"

    budget=BANKROLL*(RISK_PCT/100)
    need=len(coverage)*MIN_CHIP
    if signal.startswith("GREEN") and need>budget:
        signal="YELLOW"
        reasons.append(f"stake mínima R$ {need:.2f} > orçamento R$ {budget:.2f}")

    return {
        "signal":signal,
        "quality":quality,
        "threshold":threshold,
        "reason":" | ".join(reasons) if reasons else "todos os filtros de precisão atendidos",
        "centers":centers,
        "zones":zones,
        "coverage":coverage,
        "center_details":details,
        "coverage_probability":covp,
        "coverage_baseline":covbase,
        "coverage_lift":covlift,
        "top5_lift":top5lift,
        "window_agreement":agreement,
        "model_consensus":model_consensus,
        "entropy":entropy,
        "regime":regime,
        "calibration":calibration,
        "stake_required":round(need,2),
        "stake_budget":round(budget,2),
    }

def apply_professional_gate(sig,edge,tracker_metrics,ai_metrics,puller_metrics):
    out=dict(sig)
    original=str(out.get("signal","RED"))
    reasons=[]
    ge=(edge or {}).get("green",{}) or {}
    mature=int((tracker_metrics or {}).get("mature_models",0))
    beating=int((tracker_metrics or {}).get("models_beating_uniform",0))
    n=int(ge.get("n",0))
    z=float(ge.get("z_score",0.0))

    grade=0.0
    grade+=25*clamp(float(out.get("model_consensus",0)),0,1)
    grade+=20*clamp(float(out.get("window_agreement",0))/0.70,0,1)
    grade+=15*clamp(float(ai_metrics.get("professional_skill",0)),0,1)
    grade+=10*clamp(float(puller_metrics.get("skill",0)),0,1)
    grade+=15*clamp(beating/max(1,len(MODEL_PRIORS)),0,1)
    grade+=15*(clamp((z+1)/2,0,1) if n>=PRO_EDGE_MIN_GREEN_N else 0.5)
    grade=round(clamp(grade,0,100),1)

    if original.startswith("GREEN"):
        if n>=PRO_EDGE_MIN_GREEN_N and z<=PRO_EDGE_RED_Z:
            out["signal"]="RED"
            reasons.append(f"bloqueio OOS: Greens abaixo do esperado (z={z:.2f})")
        elif n>=PRO_EDGE_MIN_GREEN_N and z<PRO_EDGE_YELLOW_Z:
            out["signal"]="YELLOW"
            reasons.append(f"Green rebaixado por evidência OOS fraca (z={z:.2f})")
        elif original=="GREEN_ELITE" and (n<PRO_EDGE_MIN_GREEN_N or z<PRO_ELITE_MIN_Z):
            out["signal"]="GREEN_PLUS" if float(out.get("quality",0))>=GREEN_PLUS_SCORE else "GREEN"
            reasons.append("Elite rebaixado: evidência OOS ainda insuficiente")

        if (str(out.get("signal","")).startswith("GREEN")
            and mature>=PRO_MIN_TRACKED_MODELS
            and beating<PRO_MIN_MODELS_BEATING_UNIFORM):
            out["signal"]="YELLOW"
            reasons.append(f"modelos maduros sem vantagem validada ({beating}/{mature})")

    decision="APROVADA" if str(out.get("signal","")).startswith("GREEN") else \
             "OBSERVAR" if out.get("signal")=="YELLOW" else "REJEITADA"

    if reasons:
        out["reason"]=(str(out.get("reason") or "")+" | "+" | ".join(reasons)).strip(" |")

    report={
        "decision":decision,"grade":grade,"original_signal":original,
        "final_signal":out.get("signal"),"downgraded":out.get("signal")!=original,
        "reasons":reasons,"green_evidence":ge,
        "mature_models":mature,"models_beating_uniform":beating,
        "ai_professional_skill_pct":100*float(ai_metrics.get("professional_skill",0)),
        "puller_skill_pct":100*float(puller_metrics.get("skill",0)),
        "note":"Nota profissional é evidência/qualidade, não probabilidade de ganhar."
    }
    return out,report

COMMANDER_FEATURES=[
    "quality","model_consensus","temporal","coverage_lift","top5_lift",

class AICommander:
    """
    Meta-modelo que dá a decisão final da rodada.

    Os sete modelos geram evidências e probabilidades. O Commander aprende
    se a COBERTURA escolhida acertou ou não. A decisão GREEN/YELLOW/RED é
    feita aqui, e não pelo gate manual antigo.

    Modelo:
        logit(P(hit)) = logit(P_base_da_cobertura) + w·x

    Assim, com pesos zerados ele começa exatamente na chance-base e precisa
    aprender edge real para ficar mais confiante.
    """
    KEY="ai_commander_v4"

    def __init__(self,db):
        self.db=db
        st=db.get_state(self.KEY,{}) or {}
        w=st.get("weights")
        g=st.get("grad_sq")
        if not isinstance(w,list) or len(w)!=len(COMMANDER_FEATURES):
            w=[0.0]*len(COMMANDER_FEATURES)
        if not isinstance(g,list) or len(g)!=len(COMMANDER_FEATURES):
            g=[0.0]*len(COMMANDER_FEATURES)

        self.weights=[float(x) for x in w]
        self.grad_sq=[max(0.0,float(x)) for x in g]
        self.samples=int(st.get("samples",0))
        self.loss=float(st.get("loss",0.0))
        self.base_loss=float(st.get("base_loss",0.0))
        self.brier_sum=float(st.get("brier_sum",0.0))
        self.hits=int(st.get("hits",0))
        self.fast_loss=st.get("fast_loss")
        self.slow_loss=st.get("slow_loss")
        self.fast_edge=st.get("fast_edge")
        self.slow_edge=st.get("slow_edge")
        self.edge_history=[
            float(x) for x in (st.get("edge_history") or [])
            if isinstance(x,(int,float))
        ][-COMMANDER_EDGE_HISTORY:]
        self.last_decision_round=st.get("last_decision_round")
        self.lock=threading.RLock()

        if self.samples==0:
            try:
                n=self.bootstrap_from_db(COMMANDER_BOOTSTRAP_LIMIT)
                if n:
                    log.info("AI Commander warm-start: %s previsões resolvidas",n)
            except Exception:
                log.exception("Falha no warm-start do AI Commander")

    @staticmethod
    def _ema(old,value,alpha):
        return float(value) if old is None else (1-alpha)*float(old)+alpha*float(value)

    @staticmethod
    def _sigmoid(z):
        z=clamp(float(z),-30,30)
        return 1.0/(1.0+math.exp(-z))

    @staticmethod
    def _logit(p):
        p=clamp(float(p),1e-6,1-1e-6)
        return math.log(p/(1-p))

    def save(self):
        self.db.set_state(self.KEY,{
            "weights":self.weights,
            "grad_sq":self.grad_sq,
            "samples":self.samples,
            "loss":self.loss,
            "base_loss":self.base_loss,
            "brier_sum":self.brier_sum,
            "hits":self.hits,
            "fast_loss":self.fast_loss,
            "slow_loss":self.slow_loss,
            "fast_edge":self.fast_edge,
            "slow_edge":self.slow_edge,
            "edge_history":self.edge_history[-COMMANDER_EDGE_HISTORY:],
            "last_decision_round":self.last_decision_round,
        })

    def _features_from_stored(self,row):
        try: ra=json.loads(row.get("round_analysis_json") or "{}")
        except Exception: ra={}
        try: am=json.loads(row.get("ai_metrics_json") or "{}")
        except Exception: am={}
        try: cov=json.loads(row.get("coverage_json") or "[]")
        except Exception: cov=[]

        final=ra.get("final",{}) or {}
        local=ra.get("local_validation",{}) or {}
        weights=ra.get("weights",{}) or {}
        professional=ra.get("professional",{}) or {}
        ge=professional.get("green_evidence",{}) or {}
        regime=final.get("regime",{}) or {}

        quality=clamp(float(row.get("quality") or 0)/100,0,1)
        model_cons=clamp(float(final.get("model_consensus_pct",0))/100,0,1)
        temporal=clamp(float(final.get("temporal_consensus_pct",0))/100,0,1)
        cov_lift=float(final.get("coverage_lift",1.0) or 1.0)

        x=[
            quality,
            model_cons,
            temporal,
            clamp((cov_lift-1.0)/.25,-1,1),
            0.0,
            clamp(float(am.get("professional_skill",am.get("skill",0)) or 0),0,1),
            clamp(float(am.get("puller_skill",0) or 0),0,1),
            clamp(float(local.get("skill",0) or 0),0,1),
            1-clamp(float(regime.get("tv",0) or 0)/max(REGIME_BLOCK_TV,1e-6),0,1),
            0.5,
            clamp(float(ge.get("z_score",0) or 0)/2,-1,1),
            0.5,
            clamp((quality-.5)*2,0,1),
            clamp(float(weights.get("ai_alignment_pct",0) or 0)/100,0,1),
        ]
        baseline=clamp(len(set(cov))/37,1e-5,1-1e-5)
        return x,baseline

    def bootstrap_from_db(self,limit):
        with self.db.lock,self.db.conn() as c:
            rows=c.execute("""
                SELECT quality,coverage_json,ai_metrics_json,round_analysis_json,hit
                FROM predictions
                WHERE resolved_at IS NOT NULL AND hit IS NOT NULL
                ORDER BY rowid ASC LIMIT ?
            """,(int(limit),)).fetchall()

        n=0
        for rr in rows:
            row=dict(rr)
            x,base=self._features_from_stored(row)
            self._learn_vector(x,base,int(row["hit"]),save=False)
            n+=1

        if n:
            self.save()
        return n

    def build_features(self,sig,round_meta,ai_metrics,puller_metrics,edge,tracker_metrics,calibration):
        local=(round_meta or {}).get("local_puller",{}) or {}
        regime=sig.get("regime",{}) or {}
        models=(tracker_metrics or {}).get("models",{}) or {}
        mature=int((tracker_metrics or {}).get("mature_models",0))
        beating=int((tracker_metrics or {}).get("models_beating_uniform",0))
        strength=(beating/mature) if mature else 0.5
        ge=(edge or {}).get("green",{}) or {}
        ece=float((calibration or {}).get("ece",0) or 0)
        gap=float((calibration or {}).get("gap_pp",0) or 0)
        calibration_score=clamp(1-ece/20,0,1)*clamp((gap+10)/15,0,1)

        return [
            clamp(float(sig.get("quality",0))/100,0,1),
            clamp(float(sig.get("model_consensus",0)),0,1),
            clamp(float(sig.get("window_agreement",0))/0.70,0,1),
            clamp((float(sig.get("coverage_lift",1))-1)/.25,-1,1),
            clamp((float(sig.get("top5_lift",1))-1)/.35,-1,1),
            clamp(float(ai_metrics.get("professional_skill",0)),0,1),
            clamp(float(puller_metrics.get("skill",0)),0,1),
            clamp(float(local.get("skill",0)),0,1),
            1-clamp(float(regime.get("tv",0))/max(REGIME_BLOCK_TV,1e-6),0,1),
            clamp(strength,0,1),
            clamp(float(ge.get("z_score",0))/2,-1,1),
            calibration_score,
            clamp((1-float(sig.get("entropy",1)))/.12,0,1),
            clamp(float((round_meta or {}).get("ai_alignment",0))/0.50,0,1),
        ]

    def _raw_probability(self,x,baseline):
        delta=sum(w*v for w,v in zip(self.weights,x))
        return self._sigmoid(self._logit(baseline)+delta)

    def _cold_prior(self,x,baseline):
        # Prior de partida criado pelas evidências atuais.
        score=(
            0.90*(x[0]-.50)
            +0.70*(x[1]-.50)
            +0.55*(x[2]-.50)
            +0.50*x[3]
            +0.35*x[4]
            +0.45*(x[5]-.25)
            +0.30*(x[6]-.20)
            +0.30*(x[7]-.20)
            +0.25*(x[8]-.50)
            +0.25*(x[9]-.50)
            +0.18*x[10]
            +0.20*(x[11]-.50)
            +0.20*(x[12]-.30)
            +0.25*(x[13]-.40)
        )
        return self._sigmoid(self._logit(baseline)+0.75*score)

    def predict(self,x,baseline):
        learned=self._raw_probability(x,baseline)
        prior=self._cold_prior(x,baseline)

        # O histórico assume o comando gradualmente.
        learned_conf=clamp(self.samples/(self.samples+55.0),0,0.94)
        p=(1-learned_conf)*prior+learned_conf*learned

        return clamp(p,1e-5,1-1e-5),learned,prior,learned_conf

    def _learn_vector(self,x,baseline,y,save=True):
        y=float(y)
        pred=self._raw_probability(x,baseline)
        err=pred-y

        for j,v in enumerate(x):
            grad=err*float(v)+COMMANDER_L2*self.weights[j]
            grad=clamp(grad,-COMMANDER_GRAD_CLIP,COMMANDER_GRAD_CLIP)
            self.grad_sq[j]+=grad*grad
            step=COMMANDER_LR/math.sqrt(self.grad_sq[j]+1e-8)
            self.weights[j]=clamp(
                self.weights[j]-step*grad,
                -COMMANDER_WEIGHT_CLIP,COMMANDER_WEIGHT_CLIP
            )

        ll=-(y*math.log(pred)+(1-y)*math.log(1-pred))
        base_ll=-(y*math.log(baseline)+(1-y)*math.log(1-baseline))
        edge=pred-baseline

        self.samples+=1
        self.hits+=int(y)
        self.loss+=ll
        self.base_loss+=base_ll
        self.brier_sum+=(pred-y)**2
        self.fast_loss=self._ema(self.fast_loss,ll,COMMANDER_FAST_ALPHA)
        self.slow_loss=self._ema(self.slow_loss,ll,COMMANDER_SLOW_ALPHA)
        self.fast_edge=self._ema(self.fast_edge,edge,COMMANDER_FAST_ALPHA)
        self.slow_edge=self._ema(self.slow_edge,edge,COMMANDER_SLOW_ALPHA)

        if save and self.samples%5==0:
            self.save()

    def learn_from_prediction(self,prediction_row,actual):
        try:
            ra=json.loads(prediction_row.get("round_analysis_json") or "{}")
            commander=(ra.get("commander") or {})
            x=commander.get("features")
            baseline=commander.get("baseline_probability")
            if not isinstance(x,list) or len(x)!=len(COMMANDER_FEATURES):
                x,baseline=self._features_from_stored(prediction_row)
            baseline=clamp(float(baseline),1e-5,1-1e-5)
            coverage=json.loads(prediction_row.get("coverage_json") or "[]")
            y=int(int(actual) in set(int(n) for n in coverage))
            with self.lock:
                self._learn_vector([float(v) for v in x],baseline,y,save=True)
        except Exception:
            log.exception("AI Commander: erro de aprendizado")

    def metrics(self):
        n=self.samples
        avg=self.loss/n if n else None
        base=self.base_loss/n if n else None
        improvement=(base-avg) if (avg is not None and base is not None) else 0.0
        skill=clamp(improvement/0.055,0,1)

        return {
            "enabled":AI_COMMANDER_ENABLED,
            "samples":n,
            "hit_pct":100*self.hits/n if n else 0.0,
            "avg_log_loss":avg,
            "baseline_log_loss":base,
            "log_loss_improvement":improvement,
            "skill":skill,
            "fast_log_loss":self.fast_loss,
            "slow_log_loss":self.slow_loss,
            "fast_edge_pct":100*float(self.fast_edge or 0),
            "slow_edge_pct":100*float(self.slow_edge or 0),
            "weights":{COMMANDER_FEATURES[i]:round(w,4) for i,w in enumerate(self.weights)},
            "control_pct":100 if AI_COMMANDER_ENABLED else 0,
        }

    def decide(self,sig,round_meta,ai_metrics,puller_metrics,edge,tracker_metrics,calibration,source_round_id):
        out=dict(sig)
        x=self.build_features(
            sig,round_meta,ai_metrics,puller_metrics,edge,tracker_metrics,calibration
        )

        baseline=clamp(float(sig.get("coverage_baseline",0)),1e-5,1-1e-5)
        p,learned,prior,learned_conf=self.predict(x,baseline)
        predicted_edge=p-baseline

        metrics=self.metrics()
        skill=float(metrics.get("skill",0))
        adaptive_edge=max(
            COMMANDER_MIN_EDGE,
            quantile(self.edge_history,COMMANDER_EDGE_QUANTILE) if self.edge_history else COMMANDER_MIN_EDGE
        )

        authority=clamp(
            0.24*x[0]
            +0.18*x[1]
            +0.13*x[2]
            +0.08*clamp((x[3]+1)/2,0,1)
            +0.10*x[5]
            +0.07*x[6]
            +0.05*x[7]
            +0.05*x[8]
            +0.04*x[9]
            +0.03*x[11]
            +0.03*x[13],
            0,1
        )

        enough=self.samples>=COMMANDER_MIN_SAMPLES

        if enough:
            green=(
                predicted_edge>=adaptive_edge
                and authority>=COMMANDER_MIN_AUTHORITY
                and (skill>=0.08 or x[1]>=0.68)
            )
        else:
            # Cold-start seletivo. Permite sinais antes de esperar dezenas de
            # rodadas, mas exige evidência atual muito mais forte.
            green=(
                predicted_edge>=COMMANDER_EARLY_EDGE
                and authority>=COMMANDER_EARLY_AUTHORITY
                and x[0]>=0.78
                and x[1]>=0.62
            )

        if green:
            if authority>=0.80 and predicted_edge>=adaptive_edge+0.055 and skill>=0.35:
                signal="GREEN_ELITE"
            elif authority>=0.68 and predicted_edge>=adaptive_edge+0.025:
                signal="GREEN_PLUS"
            else:
                signal="GREEN"
        elif predicted_edge>max(-0.005,adaptive_edge*0.20) and authority>=0.34:
            signal="YELLOW"
        else:
            signal="RED"

        # Segurança operacional continua soberana. Isso não é decisão de aposta.
        need=float(sig.get("stake_required",0) or 0)
        budget=float(sig.get("stake_budget",0) or 0)
        if signal.startswith("GREEN") and budget>0 and need>budget:
            signal="YELLOW"

        out["signal"]=signal
        out["threshold"]=round(100*adaptive_edge,2)
        out["reason"]=(
            f"AI Commander {signal} • P(hit cobertura) {p*100:.1f}% "
            f"vs base {baseline*100:.1f}% • edge {predicted_edge*100:+.1f} pp "
            f"• autoridade {authority*100:.0f}% • skill {skill*100:.0f}%"
        )

        with self.lock:
            if str(source_round_id)!=str(self.last_decision_round):
                self.edge_history.append(float(predicted_edge))
                self.edge_history=self.edge_history[-COMMANDER_EDGE_HISTORY:]
                self.last_decision_round=str(source_round_id)
                self.save()

        report={
            "control_pct":100 if AI_COMMANDER_ENABLED else 0,
            "decision":"APROVADA" if signal.startswith("GREEN") else "OBSERVAR" if signal=="YELLOW" else "REJEITADA",
            "signal":signal,
            "predicted_probability":p,
            "learned_probability":learned,
            "cold_prior_probability":prior,
            "baseline_probability":baseline,
            "edge":predicted_edge,
            "adaptive_edge_required":adaptive_edge,
            "authority":authority,
            "skill":skill,
            "samples":self.samples,
            "learned_confidence":learned_conf,
            "warmup":not enough,
            "features":x,
            "feature_names":COMMANDER_FEATURES,
            "note":"Decisão final gerada pelo Meta-Comandante. Não é garantia de acerto."
        }
        return out,report

    def nums(self): return [int(x["number"]) for x in self.history]

    def parse(self,payload):
        if not isinstance(payload,list): raise ValueError("API não retornou lista JSON")
        out=[]
        for item in payload:
            try:
                data=item["data"]
                result=data["result"]
                num=int(result["outcome"]["number"])
                if 0<=num<=36:
                    lucky=[]
                    for x in result.get("luckyNumbersList",[]) or []:
                        try:
                            lucky.append({
                                "number":int(x.get("number")),
                                "roundedMultiplier":int(x.get("roundedMultiplier",0) or 0),
                            })
                        except Exception:
                            pass
                    table=data.get("table") or {}
                    out.append({
                        "id":str(item["id"]),
                        "number":num,
                        "settledAt":data.get("settledAt"),
                        "luckyNumbers":lucky,
                        "tableName":table.get("name","Mega Roulette"),
                        "gameType":data.get("gameType","megaroulette"),
                    })
            except Exception:
                continue
        return out

    def fetch(self, deep=False):
        """
        V3.5.1:
        - banco vazio/pouco histórico: bootstrap curto e imediato;
        - histórico suficiente: sync normal;
        - recuperação após queda: deep sync extenso.
        """
        current_count=self.db.count_spins()
        recovering=self.consecutive_errors>0

        bootstrap_mode=current_count<BOOTSTRAP_TARGET_SPINS
        deep_recovery=bool(deep and not bootstrap_mode)

        if bootstrap_mode:
            max_pages=BOOTSTRAP_PAGES
            self.sync_phase="BOOTSTRAP"
            self.sync_message=(
                f"Carregando histórico inicial: até {BOOTSTRAP_PAGES} páginas "
                f"({current_count}/{BOOTSTRAP_TARGET_SPINS} giros)"
            )
        elif deep_recovery or recovering:
            max_pages=MAX_BACKFILL_PAGES
            self.sync_phase="DEEP_SYNC"
            self.sync_message=f"Recuperando possíveis lacunas: até {MAX_BACKFILL_PAGES} páginas"
        else:
            max_pages=1
            self.sync_phase="LIVE"
            self.sync_message="Coleta ao vivo"

        records={}
        started=time.monotonic()
        pages_used=0
        found_known=False

        self.network_state="SYNCING"
        self.sync_pages_done=0
        self.sync_records_seen=0
        self.sync_new_spins=0

        for page in range(max_pages):
            url=url_for_page(page)
            r=self.session.get(
                url,
                timeout=(CONNECT_TIMEOUT,READ_TIMEOUT),
                headers={
                    "User-Agent":"Mozilla/5.0 MegaRouletteAI-Cloud-24x7",
                    "Accept":"application/json,text/plain,*/*",
                    "Connection":"keep-alive",
                }
            )
            r.raise_for_status()
            parsed=self.parse(r.json())
            pages_used+=1
            self.sync_pages_done=pages_used
            self.sync_records_seen+=len(parsed)

            if not parsed:
                break

            page_has_known=False
            for rec in parsed:
                if self.db.has_spin(rec["id"]):
                    page_has_known=True
                records[rec["id"]]=rec

            # No bootstrap precisamos das páginas iniciais mesmo que a page 0
            # já tenha algum round. Isso evita ficar preso em 27 giros.
            if not bootstrap_mode and max_pages>1 and page_has_known:
                found_known=True
                break

            if max_pages==1:
                break

        items=sorted(
            records.values(),
            key=lambda x:(x.get("settledAt") or "",x["id"])
        )

        new=0
        for rec in items:
            if not self.db.has_spin(rec["id"]):
                self.process_spin(rec, create_prediction=not FAST_BACKFILL_ANALYSIS)
                new+=1
                self.sync_new_spins=new

                # Assim que o primeiro giro entra, o painel já deixa de ser
                # "SEM DADOS". Não espera terminar o lote inteiro.
                if new==1:
                    self.sync_message=f"Primeiro giro recuperado; completando lote ({pages_used} pág.)"

        self.last_latency_ms=round((time.monotonic()-started)*1000)
        self.last_heartbeat_monotonic=time.monotonic()
        self.last_heartbeat_at=now_iso()
        self.last_fetch=now_iso()
        self.last_success=self.last_fetch
        self.last_error=""
        self.network_state="ONLINE"
        self.consecutive_errors=0
        self.last_backfill_pages=pages_used
        self.last_backfill_new=new

        if bootstrap_mode:
            self.bootstrap_completed=self.db.count_spins()>=BOOTSTRAP_TARGET_SPINS
            self.sync_phase="BOOTSTRAP_OK" if self.bootstrap_completed else "BOOTSTRAP_PARTIAL"
            self.sync_message=(
                f"Bootstrap {'concluído' if self.bootstrap_completed else 'parcial'}: "
                f"{self.db.count_spins()} giros no banco"
            )
        elif deep_recovery:
            self.last_deep_sync_monotonic=time.monotonic()
            self.sync_phase="DEEP_SYNC_OK"
            self.sync_message=f"Deep Sync concluído; {new} novo(s) giro(s)"
        else:
            self.sync_phase="LIVE_OK"
            self.sync_message=f"Coleta ao vivo; {new} novo(s) giro(s)"

        self.db.save_collector_state({
            "last_success":self.last_success,
            "last_fetch":self.last_fetch,
            "last_round_id":self.history[-1]["id"] if self.history else None,
            "last_number":self.history[-1]["number"] if self.history else None,
            "last_backfill_pages":pages_used,
            "last_backfill_new":new,
            "bootstrap_completed":self.bootstrap_completed,
            "sync_phase":self.sync_phase,
            "deep":deep_recovery,
            "found_known_round":found_known,
            "cloud_mode":CLOUD_MODE,
        })

        if self.history:
            self.ensure_prediction()

        return new

    def process_spin(self,rec,create_prediction=True):
        self.last_heartbeat_monotonic=time.monotonic()
        self.last_heartbeat_at=now_iso()
        self._status_cache=None

        if self.history:
            prev=self.history[-1]
            old=self.db.unresolved(prev["id"])
            if old:
                try:
                    self.commander.learn_from_prediction(old,int(rec["number"]))
                except Exception:
                    log.exception("Erro ao treinar AI Commander")
                try:
                    fm=json.loads(old["features_json"]); weights=json.loads(old["weights_json"])
                    explanation=OnlineAI.explain(fm[int(rec["number"])],weights,6)
                    self.db.resolve(prev["id"],rec["id"],int(rec["number"]),explanation)
                except Exception:
                    log.exception("Erro ao resolver previsão")
            try:
                self.model_tracker.evaluate(self.nums(),int(rec["number"]),self.ai)
            except Exception:
                log.exception("Erro ao validar ensemble adaptativo")
            try:
                self.puller.evaluate(self.nums(),int(rec["number"]))
            except Exception:
                log.exception("Erro ao validar puxadores")
            try:
                self.ai.learn(self.nums(),int(rec["number"]))
            except Exception:
                log.exception("Erro no aprendizado")

        if self.db.insert_spin(rec):
            self.history.append(rec)
            if len(self.history)>HISTORY_LIMIT:
                self.history=self.history[-HISTORY_LIMIT:]
            if create_prediction:
                self.ensure_prediction()

    def ensemble(self):
        nums=self.nums()
        aip,matrix=self.ai.predict(nums)
        am=self.ai.metrics()
        ai_ready=clamp(self.ai.samples/AI_MIN_SAMPLES,0,1)
        if self.ai.samples<AI_SKILL_MIN_SAMPLES:
            ai_skill=0.10*ai_ready
        else:
            ai_skill=float(am.get("professional_skill",0.0))

        models=independent_models(nums,aip)
        current_regime=regime_analysis(nums).get("level","WARMUP")
        weights=self.model_tracker.weights(current_regime)

        # Ajuste local da rodada para IA.
        ai_top=set(topk(aip,8))
        non_ai=[p for name,p in models.items() if name!="ia_online"]
        base_consensus=normalize([sum(p[i] for p in non_ai)/len(non_ai) for i in range(37)])
        base_top=set(topk(base_consensus,8))
        ai_alignment=len(ai_top&base_top)/max(1,len(ai_top|base_top))
        ai_local=clamp((0.20+0.80*ai_ready)*(0.25+0.75*ai_skill)*(0.55+0.90*ai_alignment),0.05,1.20)
        weights["ia_online"]*=ai_local

        # Ajuste local dos puxadores: suporte + skill global + skill específico do número atual.
        pp,p_report=puller_model(nums)
        local=source_local_validation(nums)
        pm=self.puller.metrics()
        global_pull_skill=float(pm.get("skill",0.0)) if self.puller.samples>=PULLER_SKILL_MIN_SAMPLES else 0.08*clamp(self.puller.samples/max(1,PULLER_SKILL_MIN_SAMPLES),0,1)
        local_n=int(local.get("n",0)); local_skill=float(local.get("skill",0.0))
        local_conf=clamp(local_n/max(1,ROUND_LOCAL_MIN_EVENTS),0,1)
        pull_local=clamp(float(p_report.get("reliability",0.0))*(0.20+0.80*global_pull_skill)*(0.20+0.80*local_conf*local_skill),0.03,1.20)
        weights["puxadores_bayes"]*=pull_local

        # Renormaliza e respeita tetos.
        total=sum(weights.values()) or 1.0
        weights={k:v/total for k,v in weights.items()}
        weights=cap_and_renormalize(weights,"ia_online",AI_MAX_BLEND)
        weights=cap_and_renormalize(weights,"puxadores_bayes",PULLER_MAX_BLEND)
        total=sum(weights.values()) or 1.0
        weights={k:v/total for k,v in weights.items()}

        raw=normalize([sum(weights[name]*models[name][i] for name in models) for i in range(37)])
        temp=self.db.temperature_calibration(TEMP_CALIBRATION_LIMIT,APP_VERSION)
        probs=temperature_scale(raw,temp.get("temperature",1.0))

        round_meta={
            "ai_alignment":ai_alignment,
            "ai_round_factor":ai_local,
            "local_puller":local,
            "pair_context":pair_context_report(nums),
            "cluster_context":cluster_context_report(nums),
            "model_weights":weights,
            "model_tracker":self.model_tracker.metrics(current_regime),
            "current_regime":current_regime,
            "temperature":temp,
        }
        return probs,matrix,weights.get("ia_online",0.0),ai_skill,weights.get("puxadores_bayes",0.0),global_pull_skill,p_report,round_meta

    def build_round_analysis(self,probs,sig,puller_report,round_meta,
                             ai_weight,ai_skill,puller_weight,puller_skill):
        nums=self.nums()
        source=int(nums[-1]) if nums else None
        previous=int(nums[-2]) if len(nums)>=2 else None
        local=round_meta.get("local_puller",{})
        pair=round_meta.get("pair_context",{})
        cluster=round_meta.get("cluster_context",{})
        regime=sig.get("regime",{})

        model_cons=clamp(float(sig.get("model_consensus",0.0)),0,1)
        temporal=clamp(float(sig.get("window_agreement",0.0))/0.70,0,1)
        reliability=clamp(float(puller_report.get("reliability",0.0)),0,1)
        local_skill=clamp(float(local.get("skill",0.0)),0,1)
        cov_lift=clamp((float(sig.get("coverage_lift",1.0))-1.0)/0.20,0,1)
        stability=1-clamp(float(regime.get("tv",0.0))/REGIME_BLOCK_TV,0,1)
        ai_align=clamp(float(round_meta.get("ai_alignment",0.0))/0.45,0,1)

        evidence=round(100*clamp(
            .22*model_cons+.16*temporal+.16*reliability+.15*local_skill+
            .11*cov_lift+.10*stability+.10*ai_align,0,1
        ),1)

        blockers=[]
        if int(local.get("n",0))<ROUND_LOCAL_MIN_EVENTS:
            blockers.append(f"pouco histórico local do {source}: {local.get('n',0)}/{ROUND_LOCAL_MIN_EVENTS}")
        if reliability<.35: blockers.append("puxadores do número atual com baixa confiabilidade")
        if model_cons<MIN_MODEL_CONSENSUS: blockers.append("consenso geral baixo")
        if float(sig.get("window_agreement",0.0))<MIN_WINDOW_AGREEMENT: blockers.append("janelas temporais discordando")
        if regime.get("level")=="MUDANÇA_FORTE": blockers.append("mudança forte de regime")
        if puller_weight<.01: blockers.append("puxadores sem skill validado para influenciar")

        if evidence>=ROUND_STRONG_EVIDENCE and not blockers: label="FORTE"
        elif evidence>=ROUND_GOOD_EVIDENCE and len(blockers)<=1: label="BOA"
        elif evidence>=55: label="MODERADA"
        else: label="FRACA"

        top_pullers=[]
        for item in (puller_report.get("top") or [])[:8]:
            t=int(item["target"]); zone=neighbors(t,2)
            top_pullers.append({
                **item,"zone":zone,
                "zone_probability_pct":100*mass(probs,zone),
                "in_final_centers":t in set(sig.get("centers") or []),
                "in_final_coverage":t in set(sig.get("coverage") or []),
            })

        return {
            "source_round_id":self.history[-1]["id"] if self.history else None,
            "source":source,"previous":previous,
            "evidence_score":evidence,"evidence_label":label,
            "note":"Índice de evidência estatística da rodada; não é porcentagem de chance.",
            "blockers":blockers,"top_pullers":top_pullers,
            "local_validation":local,"pair_context":pair,"cluster_context":cluster,
            "weights":{"ai_pct":100*ai_weight,"ai_skill_pct":100*ai_skill,
                       "puller_pct":100*puller_weight,"puller_global_skill_pct":100*puller_skill,
                       "ai_alignment_pct":100*float(round_meta.get("ai_alignment",0.0)),
                       "models_pct":{k:100*v for k,v in (round_meta.get("model_weights") or {}).items()}},
            "temperature":round_meta.get("temperature",{}),
            "final":{"signal":sig.get("signal"),"quality":sig.get("quality"),
                     "centers":sig.get("centers"),"coverage":sig.get("coverage"),
                     "coverage_lift":sig.get("coverage_lift"),
                     "model_consensus_pct":100*float(sig.get("model_consensus",0.0)),
                     "temporal_consensus_pct":100*float(sig.get("window_agreement",0.0)),
                     "regime":sig.get("regime")}
        }

    def loop(self):
        self.collector_running=True
        wait_seconds=COLLECT_INTERVAL

        # Na inicialização sempre tenta preencher a lacuna desde o último round salvo.
        need_deep_sync=True

        while not self.stop_event.is_set():
            self.last_heartbeat_monotonic=time.monotonic()
            self.last_heartbeat_at=now_iso()
            try:
                elapsed=(
                    time.monotonic()-self.last_deep_sync_monotonic
                    if self.last_deep_sync_monotonic
                    else DEEP_SYNC_INTERVAL+1
                )
                current_count=self.db.count_spins()
                bootstrap=current_count<BOOTSTRAP_TARGET_SPINS
                do_deep=(
                    (not bootstrap) and (
                        need_deep_sync
                        or self.consecutive_errors>0
                        or elapsed>=DEEP_SYNC_INTERVAL
                    )
                )

                new=self.fetch(deep=do_deep)

                if new:
                    if do_deep:
                        log.info(
                            "Cloud sync recuperou %s novo(s) giro(s) em %s página(s)",
                            new,
                            self.last_backfill_pages,
                        )
                    else:
                        log.info("Coleta recuperou %s novo(s) giro(s)",new)

                need_deep_sync=False
                wait_seconds=COLLECT_INTERVAL

            except requests.RequestException as e:
                self.consecutive_errors+=1
                self.network_state="OFFLINE"
                self.last_error=f"{type(e).__name__}: {e}"
                self.sync_phase="API_ERROR"
                self.sync_message=self.last_error

                # Ao voltar, deep sync será obrigatório.
                need_deep_sync=True

                wait_seconds=min(
                    BACKOFF_MAX,
                    max(
                        COLLECT_INTERVAL,
                        COLLECT_INTERVAL*(2**min(self.consecutive_errors-1,4))
                    )
                )

                self.db.save_collector_state({
                    "last_success":self.last_success,
                    "last_error":self.last_error,
                    "network_state":self.network_state,
                    "consecutive_errors":self.consecutive_errors,
                    "next_retry_seconds":wait_seconds,
                    "cloud_mode":CLOUD_MODE,
                })

                if self.consecutive_errors in (1,2,3) or self.consecutive_errors%10==0:
                    log.warning(
                        "API indisponível (%s). Tentativa %s; nova tentativa em %ss. "
                        "O banco permanece preservado.",
                        type(e).__name__,
                        self.consecutive_errors,
                        wait_seconds,
                    )

            except Exception as e:
                self.consecutive_errors+=1
                self.network_state="ERRO"
                self.last_error=f"{type(e).__name__}: {e}"
                self.sync_phase="ENGINE_ERROR"
                self.sync_message=self.last_error
                need_deep_sync=True
                wait_seconds=min(BACKOFF_MAX,30)
                log.exception("Erro inesperado na coleta cloud")

            self.stop_event.wait(wait_seconds)

        self.collector_running=False

def auth_ok():
    if not PANEL_USER or not PANEL_PASSWORD: return True
    h=request.headers.get("Authorization","")
    if not h.startswith("Basic "): return False
    try:
        raw=base64.b64decode(h.split(" ",1)[1]).decode()
        u,p=raw.split(":",1)
        return u==PANEL_USER and p==PANEL_PASSWORD
    except Exception:
        return False

@app.before_request
def check_auth():
    if auth_ok(): return None
    return Response("Login necessário",401,{"WWW-Authenticate":'Basic realm="Clean AI"'})

def collector_supervisor():
    """
    Se o loop do collector terminar por qualquer falha inesperada,
    reinicia automaticamente enquanto o processo cloud estiver vivo.
    """
    while not engine.stop_event.is_set():
        try:
            engine.loop()
        except Exception:
            engine.collector_restarts+=1
            log.exception(
                "Collector encerrou inesperadamente. Reiniciando em %ss.",
                COLLECTOR_RESTART_DELAY
            )
            engine.stop_event.wait(COLLECTOR_RESTART_DELAY)

        if not engine.stop_event.is_set():
            engine.collector_restarts+=1
            engine.stop_event.wait(COLLECTOR_RESTART_DELAY)



def collector_watchdog_loop():
    """
    Observa o heartbeat. Requests possui timeouts, então uma coleta não deve
    permanecer bloqueada indefinidamente. O watchdog registra qualquer atraso
    anormal e força o próximo ciclo a fazer Deep Sync.
    """
    while not engine.stop_event.is_set():
        age=time.monotonic()-engine.last_heartbeat_monotonic
        if RUN_COLLECTOR and age>COLLECTOR_HEARTBEAT_TIMEOUT:
            log.error(
                "Collector heartbeat atrasado %.1fs. Marcando rede como STALLED "
                "e forçando recuperação no próximo ciclo.",age
            )
            engine.network_state="STALLED"
            engine.last_error=f"collector heartbeat stale {age:.0f}s"
            engine.consecutive_errors=max(1,engine.consecutive_errors)
            engine._status_cache=None
        if engine.stop_event.wait(30):
            break

def backup_loop():
    # Primeiro backup alguns minutos após subir, depois periodicamente.
    if engine.stop_event.wait(180):
        return

    while not engine.stop_event.is_set():
        try:
            path=db.backup()
            log.info("Backup automático criado: %s",path)
        except Exception:
            log.exception("Falha no backup automático")

        if engine.stop_event.wait(BACKUP_INTERVAL):
            break


def database_maintenance_loop():
    last_check=0.0
    last_checkpoint=0.0

    while not engine.stop_event.is_set():
        now=time.monotonic()

        if now-last_checkpoint>=WAL_CHECKPOINT_INTERVAL:
            try:
                db.checkpoint("PASSIVE")
            except Exception as exc:
                log.warning("Checkpoint WAL falhou: %s",exc)
            last_checkpoint=now

        if now-last_check>=DB_CHECK_INTERVAL:
            try:
                status=db.integrity()
                if status!="ok":
                    log.error("Quick check SQLite retornou: %s",status)
                    if DB_AUTO_RECOVER:
                        db.recover_if_needed(force=True)
                        db.init()
            except Exception:
                log.exception("Falha na manutenção SQLite")
            last_check=now

        if engine.stop_event.wait(30):
            break


if RUN_COLLECTOR:
